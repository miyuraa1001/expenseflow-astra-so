/**
 * ==============================================================================
 * ExpenseFlow Astra SO 2021 — Google Apps Script Backend (kode.gs)
 * ==============================================================================
 * Backend REST API Sinkronisasi Realtime Master COA & Transaksi Opex Astra SO
 * Terhubung langsung ke Google Spreadsheet ID: 1wLDVXInlhwaaFr9Hs7uIN2VuPlwIkKED2G9ejFn7Tik
 */

const SECRET_TOKEN = "astra-secret-token-2026";
const SPREADSHEET_ID = "1wLDVXInlhwaaFr9Hs7uIN2VuPlwIkKED2G9ejFn7Tik"; 
const MASTER_COA_SHEET = "COA OPEX 2021 Presisi Full"; 
const TRANSACTION_SHEET = "Transaksi_Beban";

function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

function getSpreadsheet() {
  if (SPREADSHEET_ID && SPREADSHEET_ID.trim() !== "") {
    return SpreadsheetApp.openById(SPREADSHEET_ID.trim());
  }
  return SpreadsheetApp.getActiveSpreadsheet();
}

function getOrCreateTransactionSheet() {
  const ss = getSpreadsheet();
  let sheet = ss.getSheetByName(TRANSACTION_SHEET);
  if (!sheet) {
    sheet = ss.insertSheet(TRANSACTION_SHEET);
    const headers = [
      "Timestamp", "ID_Transaksi", "Tanggal", "No_Ref", "Kode_COA",
      "Nama_Akun", "Kategori_COA", "Cost_Center", "Keterangan",
      "Nominal", "Status_Pajak", "Sumber_Dana", "Operator"
    ];
    sheet.appendRow(headers);
    const headerRange = sheet.getRange(1, 1, 1, headers.length);
    headerRange.setBackground("#1e3a8a");
    headerRange.setFontColor("#ffffff");
    headerRange.setFontWeight("bold");
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function getMasterCOAList() {
  const ss = getSpreadsheet();
  const sheet = ss.getSheetByName(MASTER_COA_SHEET);
  if (!sheet) return [];

  // PENTING: getDisplayValues() mempertahankan format asli teks (titik, digit nol di depan)
  const values = sheet.getDataRange().getDisplayValues();
  if (values.length <= 1) return [];

  const coaList = [];
  let currentBabCode = "";
  let currentBabName = "";
  let currentSubBabCode = "";
  let currentSubBabName = "";

  for (let i = 1; i < values.length; i++) {
    const row = values[i];
    const isRowEmpty = row.every(cell => !cell || String(cell).trim() === "");
    if (isRowEmpty) continue;

    const rawBabCode = String(row[0] || "").trim();
    const rawBabName = String(row[1] || "").trim();
    const rawSubBabCode = String(row[2] || "").trim();
    const rawSubBabName = String(row[3] || "").trim();
    const rawSubSubBabCode = String(row[4] || "").trim();
    const rawSubSubBabName = String(row[5] || "").trim();
    const detailPenjelasan = String(row[6] || "").trim();
    const contohRedaksi = String(row[7] || "").trim();
    const statusPajak = String(row[8] || "-").trim();
    const catatanPosting = String(row[9] || "").trim();

    if (rawBabCode.toLowerCase() === "kode bab" || rawSubBabCode.toLowerCase() === "kode sub-bab") {
      continue;
    }

    if (rawBabCode && rawBabCode !== "-") {
      currentBabCode = rawBabCode;
      currentBabName = rawBabName || currentBabName;
      currentSubBabCode = "";
      currentSubBabName = "";
    }

    if (rawSubBabCode && rawSubBabCode !== "-") {
      currentSubBabCode = rawSubBabCode;
      currentSubBabName = rawSubBabName || currentSubBabName;
    }

    let kodeCOA = "";
    let namaAkun = "";

    if (rawSubSubBabCode && rawSubSubBabCode !== "-") {
      kodeCOA = rawSubSubBabCode;
      namaAkun = rawSubSubBabName || currentSubBabName || currentBabName;
    } else if (rawSubBabCode && rawSubBabCode !== "-") {
      kodeCOA = rawSubBabCode;
      namaAkun = rawSubBabName || currentBabName;
    } else if (rawBabCode && rawBabCode !== "-") {
      kodeCOA = rawBabCode;
      namaAkun = rawBabName;
    } else if (currentSubBabCode) {
      kodeCOA = currentSubBabCode;
      namaAkun = currentSubBabName;
    } else if (currentBabCode) {
      kodeCOA = currentBabCode;
      namaAkun = currentBabName;
    }

    if (!kodeCOA) continue;

    coaList.push({
      kodeCOA: kodeCOA,
      namaAkun: namaAkun,
      kodeBab: currentBabCode || rawBabCode,
      kategoriBab: currentBabName || rawBabName,
      kodeSubBab: currentSubBabCode || rawSubBabCode,
      namaSubBab: currentSubBabName || rawSubBabName,
      kodeSubSubBab: rawSubSubBabCode,
      namaSubSubBab: rawSubSubBabName,
      detailPenjelasan: detailPenjelasan,
      contohRedaksi: contohRedaksi,
      statusPajak: statusPajak || "-",
      catatanPosting: catatanPosting
    });
  }

  return coaList;
}

function doGet(e) {
  try {
    const params = e ? e.parameter : {};
    const action = params.action;

    if (action === "GET_MASTER_COA" || action === "GET_COA") {
      const coaData = getMasterCOAList();
      return createJsonResponse({
        status: "success",
        total: coaData.length,
        data: coaData
      });
    }

    if (action === "GET_TRANSACTIONS") {
      const ss = getSpreadsheet();
      const sheet = ss.getSheetByName(TRANSACTION_SHEET);
      if (!sheet) return createJsonResponse({ status: "success", total: 0, data: [] });

      const values = sheet.getDataRange().getValues();
      if (values.length <= 1) return createJsonResponse({ status: "success", total: 0, data: [] });

      const transactions = [];
      for (let i = 1; i < values.length; i++) {
        const row = values[i];
        if (row[1]) {
          transactions.push({
            id: String(row[1]),
            date: row[2] instanceof Date ? Utilities.formatDate(row[2], "GMT+7", "yyyy-MM-dd") : String(row[2] || ""),
            ref: String(row[3] || ""),
            coa: String(row[4] || ""),
            coaName: String(row[5] || ""),
            category: String(row[6] || ""),
            costCenter: String(row[7] || "CC-720 General Admin"),
            desc: String(row[8] || ""),
            amount: Number(row[9] || 0),
            taxStatus: String(row[10] || "-"),
            source: String(row[11] || "Petty Cash"),
            operator: String(row[12] || ""),
            status: "POSTED"
          });
        }
      }
      transactions.reverse();
      return createJsonResponse({ status: "success", total: transactions.length, data: transactions });
    }

    const coaCount = getMasterCOAList().length;
    return createJsonResponse({ 
      status: "online", 
      system: "ExpenseFlow Astra SO 2021 Backend", 
      version: "2.5-RealtimeSync",
      totalMasterCOA: coaCount,
      endpoints: {
        getCOA: "?action=GET_MASTER_COA",
        getTransactions: "?action=GET_TRANSACTIONS"
      },
      timestamp: new Date().toISOString()
    });
  } catch (err) {
    return createJsonResponse({ status: "error", message: "Server Error: " + err.toString() });
  }
}

function doPost(e) {
  try {
    if (!e || !e.postData || !e.postData.contents) {
      return createJsonResponse({ status: "error", message: "No payload provided" });
    }
    const postData = JSON.parse(e.postData.contents);
    if (!postData.token || postData.token !== SECRET_TOKEN) {
      return createJsonResponse({ status: "error", message: "Akses ditolak: Token autentikasi tidak valid!" });
    }

    if (postData.action === "ADD_TRANSACTION") {
      const item = postData.payload || {};
      const sheet = getOrCreateTransactionSheet();

      let finalCoaName = item.coaName || "";
      let finalCategory = item.category || "";
      let finalTaxStatus = item.taxStatus || "-";

      if ((!finalCoaName || !finalCategory) && item.coa) {
        const masterList = getMasterCOAList();
        const matched = masterList.find(c => c.kodeCOA === String(item.coa).trim());
        if (matched) {
          finalCoaName = finalCoaName || matched.namaAkun;
          finalCategory = finalCategory || matched.kategoriBab;
          finalTaxStatus = matched.statusPajak;
        }
      }

      sheet.appendRow([
        new Date(),
        item.id || Utilities.getUuid(),
        item.date || "",
        item.ref || "",
        String(item.coa || ""),
        finalCoaName,
        finalCategory,
        item.costCenter || "CC-720 General Admin",
        item.desc || "",
        Number(item.amount || 0),
        finalTaxStatus,
        item.source || "Petty Cash",
        item.operator || "Operator SO"
      ]);

      return createJsonResponse({ status: "success", message: "Transaksi berhasil dicatat ke Google Sheets!" });
    }

    return createJsonResponse({ status: "error", message: "Aksi tidak dikenali: " + postData.action });
  } catch (err) {
    return createJsonResponse({ status: "error", message: "Gagal memproses POST: " + err.toString() });
  }
}
